const express = require("express");
const mongoose = require("mongoose");
const Book = require("./models/Books.js"); 

const PORT = 5000;
const app = express();

app.use(express.json());

mongoose
  .connect("mongodb://localhost:27017/library")
  .then(() => {
    console.log("Успешное подключение к библиотеке");
  })
  .catch((error) => {
    console.log("Ошибка при подключении к БД: " + error);
  });

app.get("/books", async (req, res) => {
  try {
    const books = await Book.find();
    res.json(books);
    console.log("Список книг успешно получен");
  } catch (error) {
    console.log("Ошибка при получении книг: " + error);
  }
});

app.post("/addbook", async (req, res) => {
  try {
    const { title, description, price } = req.body;
    const newBook = await Book.create({
      title,
      description,
      price,
    });
    await newBook.save();
    res.send("Книга успешно добавлена");
  } catch (error) {
    console.log("Ошибка при добавлении книги: " + error);
  }
});

app.patch("/book/:id", async (req, res) => {
    const { id } = req.params;
    const { title, description, price } = req.body;
    
    const updatedBook = await Book.findByIdAndUpdate(
      id, 
      { title, description, price },
      { new: true } 
    );

    if (!updatedBook) {
      return res.status(404).send("Такой книги не существует");
    }
    
    res.send("Данные книги успешно изменены");
});

app.put("/book/:id", async (req, res) => {
  try {
    const { id } = req.params;
    const { title, description, price } = req.body;
    
    const updatedBook = await Book.findByIdAndUpdate(
      id, 
      { title, description, price },
      { new: true }
    );

    if (!updatedBook) {
      res.send("Такой книги не существует!");
    }

    res.send("Книга успешно перезаписана");
  } catch (error) {
    console.log("Ошибка при перезаписи книги: " + error);
    res.status(500).send("Ошибка при изменении книги");
  }
});

app.delete("/deletebook/:id", async (req, res) => {
  try {
    const { id } = req.params;
    const deletedBook = await Book.findByIdAndDelete(id);
    
    if (!deletedBook) {
      return res.status(404).send("Книга для удаления не найдена");
    }

    res.send("Книга успешно удалена");
  } catch (error) {
    console.log("Ошибка при удалении книги: " + error);
    res.status(500).send("Ошибка при удалении книги");
  }
});

app.listen(PORT, () => {
  console.log("Сервер библиотеки успешно запущен на порту: " + PORT);
});
